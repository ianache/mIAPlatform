**src/orchestrator/src/engine/dagParser.ts:**
```typescript
import { DirectedGraph } from 'graphology';
import { topologicalSort } from 'graphology-dag';

export interface FlowNode {
  id: string; type: string; node_type_id: string;
  data: { label: string; config: Record<string, unknown> };
}
export interface FlowEdge { id: string; source: string; target: string; }
export interface FlowGraph {
  nodes: FlowNode[]; edges: FlowEdge[]; misc_nodes?: unknown[];
}

export function parseAndSort(graph: FlowGraph): string[] {
  const g = new DirectedGraph();
  // Only add nodes that participate in edges (filter misc_nodes implicitly by only using graph.nodes)
  for (const node of graph.nodes) {
    g.addNode(node.id, node);
  }
  for (const edge of graph.edges) {
    // Skip self-loops and edges referencing non-existent nodes
    if (edge.source === edge.target) continue;
    if (!g.hasNode(edge.source) || !g.hasNode(edge.target)) continue;
    g.addEdge(edge.source, edge.target);
  }
  try {
    return topologicalSort(g);
  } catch (err) {
    throw new Error(`Flow contains a cycle — cannot execute: ${String(err)}`);
  }
}
```

import ivm from 'isolated-vm';

export async function runUserCode(
  code: string,
  input: unknown,
  config: Record<string, unknown>
): Promise<unknown> {
  const isolate = new ivm.Isolate({ memoryLimit: 64 });
  try {
    const context = await isolate.createContext();
    const jail = context.global;
    await jail.set('_input', new ivm.ExternalCopy(input).copyInto());
    await jail.set('_config', new ivm.ExternalCopy(config).copyInto());
    const wrappedCode = `
      ${code}
      // code MUST define: function process(input, config) { return ...; }
      JSON.stringify(process(_input, _config));
    `;
    const script = await isolate.compileScript(wrappedCode);
    const resultJson = await script.run(context, { timeout: 5000 });
    return JSON.parse(resultJson as string);
  } finally {
    isolate.dispose();  // ALWAYS dispose — prevents memory leaks
  }
}
```

**src/orchestrator/src/engine/executor.ts:**

The executor is the main loop. It:
1. Loads flow graph from mia.flows via pg pool (SELECT graph FROM mia.flows WHERE id=$1)
2. Loads all node_types referenced by flow nodes: SELECT id, code, name FROM mia.node_types WHERE id = ANY($1)
3. Calls parseAndSort(graph) — on CycleError: updateRunStatus(runId, 'failed') + emitRunEvent + return
4. Initializes PipeObject: { metadata: {flowId, runId, nodeId:'', tenantId}, payload: {}, state: {} }
5. Walks sorted node IDs in order:
   - For each node: look up nodeType from loaded map
   - Determine handler by node.type: 'source' → import sourceFile handler (stub for now), 'processor' → import processorJs handler (stub for now)
   - Execute handler: result = await handler(node, pipeObj, nodeType)
   - Deep-clone result before fan-out: use structuredClone(result) for each branch
   - Build nodeEvent: { node_id, node_name, status:'success'|'failed', duration_ms, input: pipeObj.payload, output: result, timestamp }
   - Call appendRunLog(runId, nodeEvent)
   - Call emitRunEvent(runId, { type:'node_complete', ...nodeEvent })
   - If handler throws: set nodeEvent.status='failed', nodeEvent.error=err.message, appendRunLog, emitRunEvent, call updateRunStatus(runId, 'failed', new Date()), re-throw to fail the job
6. On success: call updateRunStatus(runId, 'success', new Date(), lastPipeObj.payload)
7. Call emitRunEvent(runId, { type:'run_complete', runId, finalOutput: lastPipeObj.payload })

For parallel fan-out (node has multiple outgoing edges): use Promise.all with structuredClone(pipeObj) per branch.

import { Worker } from 'bullmq';
import { redisConnection } from '../queues.js';
import { executeFlow } from '../engine/executor.js';

export function createFlowWorker() {
  const worker = new Worker(
    'execute_flow',
    async (job) => {
      const { flowId, tenantId, runId } = job.data;
      await executeFlow(flowId, tenantId, runId);
    },
    { connection: redisConnection, concurrency: 5 }
  );
  worker.on('completed', (job) => console.log(`[worker] completed runId=${job.data.runId}`));
  worker.on('failed', (job, err) => console.error(`[worker] failed runId=${job?.data?.runId}`, err.message));
  return worker;
}
```
